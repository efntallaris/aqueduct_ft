\begin{algorithm}[H]
\small
\caption{Recipient Leader -- Handling Migrated Data}
\label{alg:recipient-leader}
\begin{algorithmic}[1]

\Upon{\fn{Receive} \msg{prepare}($G_i$, metadata) \textbf{from} donor\_leader}
  \State buffers $\leftarrow$ \fn{Allocate\_Buffers}($G_i$, metadata)
  \State \fn{Register\_RDMA}(buffers)
  \State \fn{Store\_Metadata}($G_i$, metadata, buffers)
  \State \fn{Send\_RPC}(donor\_leader, \msg{prepare\_ok}($G_i$, meta(buffers))) \Comment{addrs + rkeys}
\EndUpon

\Statex

\Upon{\fn{Receive} \msg{rdma\_done}($G_i$) \textbf{from} donor\_leader}

  \phasesep{Intra-recipient: replication}

  \State $\texttt{TRANSFERRED}[\texttt{self}][G_i] \leftarrow \texttt{true}$
  \State alive\_replicas $\leftarrow [\,]$
  \For{\textbf{each} replica $r$ \textbf{in} \fn{Replica\_Set}($G_i$)}
    \If{\fn{Check\_Alive}($r$)}
      \State alive\_replicas.append($r$)
    \EndIf
  \EndFor
  \State chain $\leftarrow$ \fn{Build\_Chain}(alive\_replicas)
  \State \fn{Async\_Replicate\_To}(chain, $G_i$) \Comment{chain-forward}
  \State \fn{Execute\_Update\_Index}($G_i$) \Comment{in parallel}

  \phasesep{Majority wait \& TRANSFER\_DURABLE}

  \State \textbf{wait until} majority of $\texttt{TRANSFERRED}[R_i][G_i] = \texttt{true}$
  \Statex \hspace{\algorithmicindent}\hspace{\algorithmicindent} \textbf{and} local index update of $G_i$ is done
  \State \fn{Log\_Append}(\msg{transfer\_durable}($G_i$))
  \State \fn{Log\_Commit}()

  \Statex
  \State \fn{Send\_RPC}(donor\_leader, \msg{durable}($G_i$)) \Comment{until acked}
\EndUpon

\Statex

\Upon{\fn{Log\_Replication\_Reply}(entry) \textbf{from} $R_i$}
  \If{entry.metadata $= \msg{rdma\_done}(\texttt{txn\_id}, G_i)$}
    \State $\texttt{TRANSFERRED}[R_i][G_i] \leftarrow \texttt{true}$
  \EndIf
  \State \fn{Append\_To\_Log}(entry)
  \State \fn{Reply\_Success}()
\EndUpon

\end{algorithmic}
\end{algorithm}
