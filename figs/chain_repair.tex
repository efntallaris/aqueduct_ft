% Chain repair with five recipient replicas (f = 2).
% Same grid and styles as Fig. 1: x unit is \columnwidth/252, y grows downwards.
% (a) B3 crashes mid-chain, so B4 and B5 never receive the data.
% (b) The leader tells B2, the closest live replica before the gap that holds
%     the data, to forward it to B4, the next live replica, and the chain
%     continues to B5.
\begin{tikzpicture}[x={\columnwidth/252},y=-1pt,
    font=\fontsize{7}{8}\fontfamily{phv}\selectfont,
    shard/.style={draw=ink,line width=0.5pt,fill=white,rounded corners=1.2pt,
                  minimum width=19pt,minimum height=10pt,inner sep=0pt},
    held/.style={shard,fill=irfill},
    leader/.style={shard,fill=ink,text=white,font=\fontsize{7}{8}\fontfamily{phv}\bfseries},
    dead/.style={shard,draw=lifegray,text=lifegray},
    bulk/.style={->,draw=ink,line width=1.2pt,>={Stealth[length=4pt,width=4pt]}},
    msg/.style={->,draw=ink,line width=0.6pt,>={Stealth[length=3.2pt,width=2.6pt]}},
    lost/.style={->,draw=lifegray,line width=1.2pt,>={Stealth[length=4pt,width=4pt]},
                 dash pattern=on 2.5pt off 1.5pt},
    lbl/.style={inner sep=1pt,text=ink},
    rowlbl/.style={anchor=west,inner sep=0pt,font=\fontsize{7}{8}\fontfamily{phv}\bfseries},
  ]
  \definecolor{ink}{RGB}{34,34,34}
  \definecolor{lifegray}{RGB}{165,165,165}
  \definecolor{irfill}{gray}{0.86}
  \def\Gi{\textit{G}\textsubscript{\textit{i}}}
  \def\crash#1#2{%
    \draw[ink,line width=1pt,line cap=round] (#1-2.6,#2-2.6) -- (#1+2.6,#2+2.6) (#1-2.6,#2+2.6) -- (#1+2.6,#2-2.6);}

  % replica x positions
  \def\Ba{40} \def\Bb{88} \def\Bc{136} \def\Bd{184} \def\Be{232}

  % (a) B3 crashes during chain replication
  \node[rowlbl] at (0,10) {(a)};
  \node[leader] at (\Ba,10) {B1};
  \node[held]   at (\Bb,10) {B2};
  \node[dead]   at (\Bc,10) {B3};
  \node[shard]  at (\Bd,10) {B4};
  \node[shard]  at (\Be,10) {B5};
  \crash{\Bc+10}{5}
  \draw[bulk] (\Ba+11,10) -- (\Bb-11,10);
  \draw[bulk] (\Bb+11,10) -- (\Bc-11,10);
  \draw[lost] (\Bc+11,10) -- (\Bd-11,10);
  \draw[lost] (\Bd+11,10) -- (\Be-11,10);

  % (b) B1 tells B2 to forward the data to B4
  \node[rowlbl] at (0,46) {(b)};
  \node[leader] at (\Ba,46) {B1};
  \node[held]   at (\Bb,46) {B2};
  \node[dead]   at (\Bc,46) {B3};
  \node[held]   at (\Bd,46) {B4};
  \node[held]   at (\Be,46) {B5};
  \crash{\Bc+10}{41}
  \draw[msg] (\Ba+11,46) -- node[lbl,above=1pt,font=\fontsize{6.3}{7}\fontfamily{phv}\selectfont] {forward} (\Bb-11,46);
  \draw[bulk,rounded corners=4pt] (\Bb,40) -- (\Bb,31) -- (\Bd,31) -- (\Bd,40);
  \draw[bulk] (\Bd+11,46) -- (\Be-11,46);

  % majority: the leader plus f = 2 followers that report the data
  \draw[ink,line width=0.4pt] (\Ba-10,55) -- ++(0,2.5) -- (\Bd+10,57.5) -- ++(0,-2.5);
  \node[lbl,anchor=north] at ({(\Ba+\Bd)/2},59) {majority $\Rightarrow$ TRANSFER\_DURABLE(\Gi)};

  % legend
  \node[inner sep=0pt,anchor=base] at (136,80) {%
    \tikz[baseline=-2.5pt]\node[held,minimum width=10pt,minimum height=7pt]{};\hspace{3pt}holds \Gi{}\hspace{12pt}%
    \tikz[baseline=-2.5pt]\draw[msg] (0pt,0pt) -- (14pt,0pt);\hspace{3pt}RPC\hspace{12pt}%
    \tikz[baseline=-2.5pt]\draw[bulk] (0pt,0pt) -- (14pt,0pt);\hspace{3pt}RDMA write};
\end{tikzpicture}%
