% Fig. 1 -- baseline migration protocol.
% Drawn at its final size (one IEEEtran column, 252pt wide): \input it
% directly, without \resizebox, so the text prints at 7.5pt.
% y grows downwards; x positions of the six lifelines are defined below.
\begin{tikzpicture}[x=1pt,y=-1pt,
    font=\fontsize{7.5}{9}\fontfamily{phv}\selectfont,
    >={Stealth[length=4pt,width=3.2pt]},
    lifeline/.style={draw=lifegray,line width=0.8pt},
    shard/.style={draw=ink,line width=0.6pt,fill=white,rounded corners=1.5pt,
                  minimum width=24pt,minimum height=12pt,inner sep=0pt},
    leader/.style={shard,fill=ink,text=white,font=\fontsize{7.5}{9}\fontfamily{phv}\bfseries},
    msg/.style={->,draw=ink,line width=0.6pt},
    bulk/.style={->,draw=ink,line width=1.8pt,>={Stealth[length=5pt,width=5pt]}},
    notify/.style={->,draw=ink,line width=0.6pt,dash pattern=on 2.2pt off 1.4pt},
    lbl/.style={inner sep=1pt,text=ink},
    phase/.style={rounded corners=2pt},
    phaselbl/.style={anchor=north west,inner sep=1.5pt,outer sep=0pt,
                     font=\fontsize{7.5}{9}\fontfamily{phv}\bfseries},
  ]
  \definecolor{ink}{RGB}{34,34,34}
  \definecolor{lifegray}{RGB}{165,165,165}
  \definecolor{llfill}{gray}{0.93}  % leader-to-leader phase
  \definecolor{llink}{gray}{0.13}
  \definecolor{irfill}{gray}{0.86}  % intra-recipient phase
  \definecolor{irink}{gray}{0.13}
  \def\Gi{\textit{G}\textsubscript{\textit{i}}}
  \def\us{{\fontencoding{T1}\selectfont\_}}% thin underscore glyph

  % lifeline x positions
  \def\Ao{22}  \def\At{62}  \def\Ah{104}  % donor: A1, A2, A3 (leader)
  \def\Bo{150} \def\Bt{193} \def\Bh{232}  % recipient: B1 (leader), B2, B3
  \def\bottom{336}

  % phase backgrounds
  \fill[phase,llfill] (0,65) rectangle (252,142);
  \fill[phase,irfill] (0,148) rectangle (252,315);

  % lifelines
  \foreach \x in {\Ao,\At,\Ah,\Bo,\Bt,\Bh} \draw[lifeline] (\x,33) -- (\x,\bottom);

  % replica groups and shards
  \node[font=\fontsize{7.5}{9}\fontfamily{phv}\bfseries,anchor=base] at (\At,6) {Donor replica group};
  \node[font=\fontsize{7.5}{9}\fontfamily{phv}\bfseries,anchor=base] at (\Bt,6) {Recipient replica group};
  \draw[ink,line width=0.5pt] (\Ao-12,15) -- ++(0,-3) -- (\Ah+14,12) -- ++(0,3);
  \draw[ink,line width=0.5pt] (\Bo-14,15) -- ++(0,-3) -- (\Bh+12,12) -- ++(0,3);
  \node[shard]  at (\Ao,27) {A1};
  \node[shard]  at (\At,27) {A2};
  \node[leader] at (\Ah,27) {A3};
  \node[leader] at (\Bo,27) {B1};
  \node[shard]  at (\Bt,27) {B2};
  \node[shard]  at (\Bh,27) {B3};
  \node[lbl,fill=white,font=\fontsize{7}{8}\fontfamily{phv}\itshape] at (\Ah,41) {leader};
  \node[lbl,fill=white,font=\fontsize{7}{8}\fontfamily{phv}\itshape] at (\Bo,41) {leader};

  % G_i_START: donor leader logs the migration intent before the transfer
  \node[lbl,above=2pt,fill=white] at ({(\Ao+\Ah)/2},56) {\Gi\us START};
  \draw[notify] (\Ah,56) -- (\Ao,56);
  \draw[notify] (\Ah,56) -- (\At,56);

  % phase 1: leader-to-leader
  \node[phaselbl,text=llink,fill=llfill] at (2,67) {Leader-to-leader};
  \draw[msg]  (\Ah,86) -- node[lbl,above=2pt] {prepare}  (\Bo,86);
  \draw[msg]  (\Bo,101) -- node[lbl,above=2pt] {MRs}      (\Ah,101);
  \draw[bulk] (\Ah,116) -- node[lbl,above=2pt] {transfer} (\Bo,116);
  \draw[msg]  (\Ah,130) -- node[lbl,above=2pt] {done}     (\Bo,130);
  \draw[msg]  (\Bo,135) -- (\Ah,135);

  % phase 2: intra-recipient chain replication B1 -> B2 -> B3,
  % with B1's local index update (IU) running in parallel
  \node[phaselbl,text=irink,fill=irfill] at (2,150) {Intra-recipient};
  \draw[msg,rounded corners=2pt] (\Bo,160.5) -- ++(12,0) -- ++(0,9) -- (\Bo,169.5);
  \node[lbl,right=3pt,fill=irfill] at (\Bo+12,165) {ack B1 (1/3)};
  \draw[msg]  (\Bo,184) -- node[lbl,above=2pt] {prepare}  (\Bt,184);
  \draw[msg]  (\Bt,199) -- node[lbl,above=2pt] {MRs}      (\Bo,199);
  \draw[bulk] (\Bo,214) -- node[lbl,above=2pt] {transfer} (\Bt,214);
  \draw[msg]  (\Bo,228) -- node[lbl,above=2pt] {done}     (\Bt,228);
  \draw[bulk] (\Bt,243) -- node[lbl,above=2pt] {forward}  (\Bh,243);
  \draw[msg]  (\Bt,257) -- node[lbl,above=2pt] {done}     (\Bh,257);
  \draw[msg]  (\Bt,272) -- node[lbl,above=2pt] {ack (2/3)} (\Bo,272);
  \draw[ink,line width=0.6pt,decorate,decoration={brace,amplitude=4pt}]
    (\Bo-7,276) -- node[lbl,left=5pt] {IU} (\Bo-7,156);

  % majority reached: the migrated data is durable in the recipient group
  \draw[irink,line width=1pt] (\Bo,289) -- (\Bh,289)
    node[midway,lbl,inner sep=2pt,fill=irfill,text=irink,
         font=\fontsize{7.5}{9}\fontfamily{phv}\bfseries] {majority reached};
  \node[lbl,above=2pt,fill=irfill] at ({(\Bo+\Bh)/2},309) {INDEX\us UPDATE};
  \draw[notify] (\Bo,309) -- (\Bt,309);
  \draw[notify] (\Bo,309) -- (\Bh,309);

  % G_i_DONE: recipient leader reports completion, donor leader logs it
  \node[lbl,above=2pt,fill=white] at ({(\Ao+\Ah)/2},328) {\Gi\us DONE};
  \draw[msg] (\Bo,328) -- (\Ah,328);
  \draw[notify] (\Ah,328) -- (\Ao,328);
  \draw[notify] (\Ah,328) -- (\At,328);

  % legend: one centred row, equal gaps, arrows at mid x-height of the text
  \def\key#1{\tikz[baseline=-2.7pt]\draw[#1] (0pt,0pt) -- (18pt,0pt);\hspace{4pt}}
  \node[inner sep=0pt,anchor=base] at (126,352) {%
    \key{msg}RPC\hspace{16pt}%
    \key{notify}Replicated log entry\hspace{16pt}%
    \key{bulk}RDMA write};
\end{tikzpicture}%
